```bash
python3 - <<'PY'
import json
import os
import shutil
from pathlib import Path

description = Path(
    "/mnt/weight/k3/"
    "[REDACTED_SECRET]"
    "quant_model_description.json"
)

    "quant_model_description.pre-quarot-map.json"

        "rotation_map": {
            "global_rotation": "optional/quarot.safetensors"

if data.get("optional") == expected:
    print("descriptor already correct")
    raise SystemExit(0)

if data.get("optional") not in ({}, None):
    raise SystemExit(
        f"refusing to overwrite unexpected optional={data.get('optional')!r}"
    )

needle = b'"optional": {}'
replacement = (
    b'"optional": {"quarot": {"rotation_map": '
    b'{"global_rotation": "optional/quarot.safetensors"}}}'
)

if raw.count(needle) != 1:
    raise SystemExit(
        f"expected exactly one {needle!r}, found {raw.count(needle)}"
    )

if not backup.exists():
    shutil.copy2(description, backup)

updated = raw.replace(needle, replacement, 1)
assert json.loads(updated)["optional"] == expected

tmp = description.with_name(f".{description.name}.tmp-{os.getpid()}")
tmp.write_bytes(updated)
shutil.copymode(description, tmp)
os.replace(tmp, description)

print(f"updated: {description}")
print(f"backup:  {backup}")
PY
```
